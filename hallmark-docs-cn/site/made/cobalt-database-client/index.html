leeds-pending.sql

1SELECT c.name, count(o.id) AS pending
2FROM orders o
3JOIN customers c ON c.id = o.customer_id
4WHERE o.status = 'pending'
5  AND c.city = 'Leeds'
6GROUP BY c.name
7ORDER BY pending DESC;
