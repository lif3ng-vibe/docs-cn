## 示例

### 确认操作

一个简单的确认/取消对话框，两个 DialogClose 按钮分别充当取消与确认操作。

<Example code={`<button onclick="document.getElementById('confirm').showModal()">删除项目</button>

<Dialog id="confirm">
  <DialogContent class="max-w-sm">
    <div class="p-6">
      <h3 class="text-base font-semibold mb-2">确定要继续吗？</h3>
      <p class="text-sm text-muted-foreground mb-4">此操作无法撤销。</p>
      <div class="flex justify-end gap-2">
        <DialogClose class="px-3 py-1.5 rounded-md border border-border text-sm">取消</DialogClose>
        <DialogClose class="px-3 py-1.5 rounded-md bg-danger text-white text-sm">删除</DialogClose>
      </div>
    </div>
  </DialogContent>
</Dialog>`} contentClass="w-full">
  <div class="w-full text-left flex flex-col items-center gap-3">
    <button
      type="button"
      onclick="document.getElementById('demo-confirm').showModal()"
      class="inline-flex items-center justify-center px-4 py-2 rounded-md bg-danger text-white text-sm font-medium hover:opacity-90 transition-opacity cursor-pointer"
    >
      删除项目
    </button>
    <Dialog id="demo-confirm">
      <DialogContent class="max-w-sm">
        <div class="p-6">
          <h3 class="text-base font-semibold text-foreground mb-2">确定要继续吗？</h3>
          <p class="text-sm text-muted-foreground mb-4">此操作无法撤销。</p>
          <div class="flex justify-end gap-2">
            <DialogClose class="px-3 py-1.5 rounded-md border border-border text-sm hover:bg-accent transition-colors cursor-pointer">
              取消
            </DialogClose>
            <DialogClose class="px-3 py-1.5 rounded-md bg-danger text-white text-sm hover:opacity-90 transition-opacity cursor-pointer">
              删除
            </DialogClose>
          </div>
        </div>
      </DialogContent>
    </Dialog>
  </div>
</Example>
